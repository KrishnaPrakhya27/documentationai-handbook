Returns the import run. A run left behind by an import that never reported back is returned as failed.

**GET** `/org/{organizationId}/imports/{importRunId}`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="organizationId" param-type="string" required={true}>
  The unique id of the organization
</ParamField>

<ParamField path="importRunId" param-type="string" required={true}>
  The unique id of the import run
</ParamField>

**Responses**

`200` OK

<ResponseField name="object" field-type="string" enum="import-run">
  Type of Object, always equals to "import-run"
</ResponseField>

<ResponseField name="id" field-type="string" required={true}>
  Unique ID of the import
</ResponseField>

<ResponseField name="startedAt" field-type="string" />

<ResponseField name="completedAt" field-type="string" />

<ResponseField name="createdAt" field-type="string" />

<ResponseField name="updatedAt" field-type="string" />

<ResponseField name="status" field-type="string" required={true} enum="pending,in-progress,completed,failed,cancelled">
  Status of the import
</ResponseField>

<ResponseField name="reason" field-type="string">
  Reason for the import failure when `status` is `failed`.
</ResponseField>

<ResponseField name="source" field-type="string" />

<ResponseField name="target" field-type="object">
  <Expandable title="Object Properties">
    <ResponseField name="space" field-type="string" required={true}>
      ID of the space to import in
    </ResponseField>

    <ResponseField name="changeRequest" field-type="string">
      ID of the change request to import in
    </ResponseField>

    <ResponseField name="page" field-type="string">
      ID of the page to import in
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="site" field-type="string">
  ID of the site the target space belongs to, when it belongs to exactly one.
</ResponseField>

<ResponseField name="createdBy" field-type="string">
  ID of the user who started the import.
</ResponseField>

<ResponseField name="urls" field-type="object">
  URLs associated with the object

  <Expandable title="Object Properties">
    <ResponseField name="location" field-type="string" required={true}>
      URL of the import run in the API
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="pages" field-type="array">
  Pages included in the import

  <Expandable title="Array Items">
    <ResponseField name="id" field-type="string" required={true}>
      ID of page
    </ResponseField>

    <ResponseField name="sourceURL" field-type="string">
      Source URL where the page has been imported from
    </ResponseField>

    <ResponseField name="type" field-type="string" required={true} enum="page,folder">
      Type of page
    </ResponseField>

    <ResponseField name="status" field-type="string" required={true} enum="complete,pending">
      Status of the page
    </ResponseField>
  </Expandable>
</ResponseField>

`404` Not Found

<ResponseField name="error" field-type="object" required={true}>
  <Expandable title="Object Properties">
    <ResponseField name="code" field-type="integer" required={true} enum="404" />

    <ResponseField name="message" field-type="string" required={true} />
  </Expandable>
</ResponseField>
