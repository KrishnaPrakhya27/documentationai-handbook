Lists the built-in style guide starter templates that can seed a new styleguide, including the markdown content used to render each template's preview.

**GET** `/orgs/{organizationId}/styleguide-templates`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="organizationId" param-type="string" required={true}>
  The unique id of the organization
</ParamField>

**Responses**

`200` OK

<ResponseField name="templates" field-type="array" required={true}>
  <Expandable title="Array Items">
    <ResponseField name="id" field-type="string" required={true}>
      Identifier of the template, passed to createSiteStyleguideById to seed a style guide from it.
    </ResponseField>

    <ResponseField name="title" field-type="string" required={true}>
      Human-readable name of the template.
    </ResponseField>

    <ResponseField name="subtitle" field-type="string">
      Short one-line description of the template.
    </ResponseField>

    <ResponseField name="emoji" field-type="string">
      Emoji illustrating the template.
    </ResponseField>

    <ResponseField name="markdown" field-type="string" required={true}>
      Markdown content of the template, rendered as a preview.
    </ResponseField>
  </Expandable>
</ResponseField>
