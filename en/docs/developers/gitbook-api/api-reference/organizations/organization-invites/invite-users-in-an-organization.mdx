**POST** `/orgs/{organizationId}/invites`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="organizationId" param-type="string" required={true}>
  The unique id of the organization
</ParamField>

**Request body**

<ParamField body="emails" param-type="string[]" required={true} />

<ParamField body="role" param-type="string">
  The role of a member in an organization, null for guests
</ParamField>

<ParamField body="sso" param-type="boolean">
  If true, invites the user as an SSO user of the organization. Defaults to false.
</ParamField>

**Responses**

`200` OK

<ResponseField name="users" field-type="string[]" required={true} />

<ResponseField name="invited" field-type="number" required={true}>
  The number of users who were added to the organization
</ResponseField>

<ResponseField name="failedSSOEmails" field-type="string[]" />

`400` Bad Request

    <ResponseField name="message" field-type="string" required={true} />
  </Expandable>
</ResponseField>
