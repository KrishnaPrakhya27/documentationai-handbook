**PATCH** `/orgs/{organizationId}/members/{userId}`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="organizationId" param-type="string" required={true}>
  The unique id of the organization
</ParamField>

<ParamField path="userId" param-type="string" required={true}>
  The unique ID of the User
</ParamField>

**Request body**

<ParamField body="role" param-type="string">
  The role of a member in an organization, null for guests
</ParamField>

**Responses**

`200` The member has been updated

<ResponseField name="object" field-type="string" required={true} enum="member">
  Type of Object, always equals to "member"
</ResponseField>

<ResponseField name="id" field-type="string" required={true}>
  Unique identifier for the user.
</ResponseField>

<ResponseField name="role" field-type="string" required={true}>
  The role of a member in an organization, null for guests
</ResponseField>

<ResponseField name="user" field-type="object" required={true}>

    <ResponseField name="object" field-type="string" required={true} enum="user">
      Type of Object, always equals to "user"

    <ResponseField name="id" field-type="string" required={true}>
      Unique identifier for the user
    </ResponseField>
