**GET** `/orgs/{organizationId}/translations-glossary`

**Parameters**

<ParamField header="Authorization" param-type="string" required={true} />

<ParamField path="organizationId" param-type="string" required={true}>
  The unique id of the organization
</ParamField>

<ParamField query="page" param-type="string">
  Identifier of the page results to fetch.
</ParamField>

<ParamField query="limit" param-type="number">
  The number of results per page
</ParamField>

<ParamField query="orderBy" param-type="string" enum="en,fr,de,es,it,pt,pt-br,ru,ja,zh,zh-tw,yue,ko,ar,hi,nl,pl,tr,sv,no,da,fi,el,cs,hu,ro,th,vi,id,ms,he,uk,sk,bg,hr,lt,lv,et,sl">

</ParamField>
