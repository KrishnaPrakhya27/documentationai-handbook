Define terms and specify their translations for different languages to ensure consistent wording.

## The GlossaryEntry object

<ResponseField name="object" field-type="string" required={true} enum="glossary-entry" />

<ResponseField name="id" field-type="string" required={true}>
  Unique identifier of the glossary entry
</ResponseField>

<ResponseField name="translations" field-type="object" required={true}>
  Mapping of language codes to translated terms
</ResponseField>

<ResponseField name="createdAt" field-type="string" required={true} />

<ResponseField name="updatedAt" field-type="string" required={true} />

<ResponseField name="urls" field-type="object" required={true}>
  <Expandable title="Object Properties">
    <ResponseField name="location" field-type="string" required={true}>
      URL of the glossary entry in the API
    </ResponseField>
  </Expandable>
</ResponseField>
