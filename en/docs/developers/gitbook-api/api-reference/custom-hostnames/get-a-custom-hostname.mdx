**GET** `/custom-hostnames/{hostname}`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="hostname" param-type="string" required={true}>
  The custom hostname, for example "docs.gitbook.com"
</ParamField>

**Responses**

`200` OK

<ResponseField name="object" field-type="string" required={true} enum="custom-hostname" />

<ResponseField name="hostname" field-type="string" required={true}>
  Custom hostname for the site, for e.g. docs.mycompany.com
</ResponseField>

<ResponseField name="target" field-type="string" required={true} />

<ResponseField name="isActive" field-type="boolean" required={true} />

<ResponseField name="status" field-type="string" />

<ResponseField name="urls" field-type="object" required={true}>
  URLs associated with the object

  <Expandable title="Object Properties">
    <ResponseField name="location" field-type="string" required={true}>
      URL of the custom hostname in the API
    </ResponseField>
  </Expandable>
</ResponseField>
