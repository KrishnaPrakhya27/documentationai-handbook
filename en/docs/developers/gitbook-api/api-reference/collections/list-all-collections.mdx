**GET** `/orgs/{organizationId}/collections`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="organizationId" param-type="string" required={true}>
  The unique id of the organization
</ParamField>

<ParamField query="page" param-type="string">
  Identifier of the page results to fetch.
</ParamField>

<ParamField query="limit" param-type="number">
  The number of results per page
</ParamField>

<ParamField query="nested" param-type="boolean">
  If true, all nested collections will be listed
</ParamField>

**Responses**

`200` OK

<ResponseField name="next" field-type="object">
  <Expandable title="Object Properties">
    <ResponseField name="page" field-type="string" required={true}>
      Unique identifier to query the next results page
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="count" field-type="number">
  Total count of objects in the list
</ResponseField>

<ResponseField name="items" field-type="array" required={true}>
  <Expandable title="Array Items">
    <ResponseField name="object" field-type="string" required={true} enum="collection">
      Type of Object, always equals to "collection"
    </ResponseField>

    <ResponseField name="id" field-type="string" required={true}>
      Unique identifier for the collection
    </ResponseField>

    <ResponseField name="title" field-type="string" required={true}>
      Title of the collection
    </ResponseField>

    <ResponseField name="description" field-type="string">
      Description of the collection
    </ResponseField>

    <ResponseField name="organization" field-type="string" required={true}>
      ID of the organization owning this collection
    </ResponseField>

    <ResponseField name="parent" field-type="string">
      ID of the parent collection, if any
    </ResponseField>

    <ResponseField name="defaultLevel" field-type="string" required={true}>
      Default level for a piece of content
    </ResponseField>

    <ResponseField name="urls" field-type="object" required={true}>
      URLs associated with the object

      <Expandable title="Object Properties">
        <ResponseField name="location" field-type="string" required={true}>
          URL of the collection in the API
        </ResponseField>

        <ResponseField name="app" field-type="string" required={true}>
          URL of the collection in the application
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="permissions" field-type="object" required={true}>
      The set of permissions for the collection

      <Expandable title="Object Properties">
        <ResponseField name="view" field-type="boolean" required={true}>
          Can the user view the collection.
        </ResponseField>

        <ResponseField name="admin" field-type="boolean" required={true}>
          Can the user edit the title/description.
        </ResponseField>

        <ResponseField name="viewInviteLinks" field-type="boolean" required={true}>
          Can the user view the invite links of the collection.
        </ResponseField>

        <ResponseField name="create" field-type="boolean" required={true}>
          Can the user create spaces/collections in this collection.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>
