**GET** `/collections/{collectionId}/permissions/users`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="collectionId" param-type="string" required={true}>
  The unique id of the collection
</ParamField>

<ParamField query="page" param-type="string">
  Identifier of the page results to fetch.
</ParamField>

<ParamField query="limit" param-type="number">
  The number of results per page
</ParamField>

**Responses**

`200` Listing of users who have been added to a collection.

<ResponseField name="next" field-type="object">
  <Expandable title="Object Properties">
    <ResponseField name="page" field-type="string" required={true}>
      Unique identifier to query the next results page
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="count" field-type="number">
  Total count of objects in the list
</ResponseField>

<ResponseField name="items" field-type="array" required={true}>
  <Expandable title="Array Items">
    <ResponseField name="permission" field-type="string" required={true} enum="admin,create,edit,review,comment,read">
      "The role of a member in an organization.
      "admin": Can administrate the content: create, delete spaces, ...
      "create": Can create content.
      "review": Can review content.
      "edit": Can edit the content (live or change requests).
      "comment": Can access the content and its discussions.
      "read": Can access the content, but cannot update it in any way.
    </ResponseField>

    <ResponseField name="user" field-type="object" required={true}>
      <Expandable title="Object Properties">
        <ResponseField name="object" field-type="string" required={true} enum="user">
          Type of Object, always equals to "user"
        </ResponseField>

        <ResponseField name="id" field-type="string" required={true}>
          Unique identifier for the user
        </ResponseField>

        <ResponseField name="displayName" field-type="string" required={true}>
          Full name for the user
        </ResponseField>

        <ResponseField name="email" field-type="string">
          Email address of the user
        </ResponseField>

        <ResponseField name="photoURL" field-type="string">
          URL of the user's profile picture
        </ResponseField>

        <ResponseField name="urls" field-type="object" required={true}>
          URLs associated with the object

          <Expandable title="Object Properties">
            <ResponseField name="location" field-type="string" required={true}>
              URL of the user in the API
            </ResponseField>
          </Expandable>
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

`404` Not Found
