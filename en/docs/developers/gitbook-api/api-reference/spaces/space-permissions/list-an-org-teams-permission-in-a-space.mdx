**GET** `/spaces/{spaceId}/permissions/teams`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="spaceId" param-type="string" required={true}>
  The unique id of the space
</ParamField>

<ParamField query="page" param-type="string">
  Identifier of the page results to fetch.
</ParamField>

<ParamField query="limit" param-type="number">
  The number of results per page
</ParamField>

**Responses**

`200` Listing of teams who have been added to a space.

<ResponseField name="next" field-type="object">
  <Expandable title="Object Properties">
    <ResponseField name="page" field-type="string" required={true}>
      Unique identifier to query the next results page
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="count" field-type="number">
  Total count of objects in the list
</ResponseField>

<ResponseField name="items" field-type="array" required={true}>
  <Expandable title="Array Items">
    <ResponseField name="permission" field-type="string" required={true} enum="admin,create,edit,review,comment,read">
      "The role of a member in an organization.
      "admin": Can administrate the content: create, delete spaces, ...
      "create": Can create content.
      "review": Can review content.
      "edit": Can edit the content (live or change requests).
      "comment": Can access the content and its discussions.
      "read": Can access the content, but cannot update it in any way.
    </ResponseField>

    <ResponseField name="team" field-type="object" required={true}>
      <Expandable title="Object Properties">
        <ResponseField name="object" field-type="string" required={true} enum="team">
          Type of Object, always equals to "team"
        </ResponseField>

        <ResponseField name="id" field-type="string" required={true}>
          Unique identifier for the team.
        </ResponseField>

        <ResponseField name="title" field-type="string" required={true}>
          Title of the team
        </ResponseField>

        <ResponseField name="members" field-type="integer" required={true}>
          Count of members in this team.
        </ResponseField>

        <ResponseField name="spaces" field-type="number" required={true}>
          Count of spaces this team has access to.
        </ResponseField>

        <ResponseField name="createdAt" field-type="string" required={true} />

        <ResponseField name="permissions" field-type="object" required={true}>
          The set of permissions for the team

          <Expandable title="Object Properties">
            <ResponseField name="admin" field-type="boolean" required={true}>
              Can the user manage the team
            </ResponseField>

            <ResponseField name="view" field-type="boolean" required={true}>
              Can the user view the team and list its members
            </ResponseField>
          </Expandable>
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

`404` Not Found

  <Expandable title="Object Properties">
    <ResponseField name="code" field-type="integer" required={true} enum="404" />

    <ResponseField name="message" field-type="string" required={true} />
  </Expandable>
