Server-renders a document from a provided content source definition, evaluating template expressions and resolving variables to produce a fully computed document. Use this to preview how dynamic content will look without publishing a full revision.

**POST** `/spaces/{spaceId}/content/computed/document`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="spaceId" param-type="string" required={true}>
  The unique id of the space
</ParamField>

<ParamField query="schema" param-type="string" enum="current,next">
  Version of the schema used for the document.
</ParamField>

**Request body**

<ParamField body="source" param-type="string" required={true} />

<ParamField body="seed" param-type="string" required={true}>
  Seed to use for the generation of IDs.
</ParamField>

**Responses**

`200` Document computed

<ResponseField name="object" field-type="string" required={true} enum="document" />

<ResponseField name="data" field-type="object" required={true}>
  <Expandable title="Object Properties">
    <ResponseField name="schemaVersion" field-type="integer">
      The schema version of the document. If undefined, the document is considered to be of the latest schema version.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="nodes" field-type="string[]" required={true} />

<ResponseField name="meta" field-type="object">
  <Expandable title="Object Properties">
    <ResponseField name="token" field-type="string">
      A content token that can be used to fetch the content referenced in this document from the API.
    </ResponseField>
  </Expandable>
</ResponseField>
