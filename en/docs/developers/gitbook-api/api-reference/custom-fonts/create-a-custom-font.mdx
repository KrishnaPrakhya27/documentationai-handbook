**PUT** `/orgs/{organizationId}/fonts`

**Parameters**

<ParamField header="Authorization" param-type="string" required={true} />

<ParamField path="organizationId" param-type="string" required={true}>
  The unique id of the organization
</ParamField>

**Request body**

<ParamField body="fontFamily" param-type="string" required={true}>
  The human-readable font-family name used in CSS (e.g., "Open Sans", "Playfair Display").
</ParamField>

<ParamField body="fontFaces" param-type="array" required={true}>
  Array of font faces specifying weights and their corresponding storage keys.
  At least one font face with weight 400 (regular) must be provided.

  <Expandable title="Array Items">
    <ParamField body="weight" param-type="integer" required={true}>
      Numeric representation of the font weight (400=regular, 500=medium, 700=bold, 900=black).
    </ParamField>

    <ParamField body="storageKey" param-type="string" required={true}>
      The path of the file in the storage bucket
    </ParamField>
  </Expandable>
</ParamField>

**Responses**

`201` Custom font created successfully.

<ResponseField name="id" field-type="string" required={true}>
  A globally unique identifier for the font definition.
</ResponseField>

<ResponseField name="custom" field-type="boolean" required={true}>
  Whether the font is a custom font. If false, this font is provided by GitBook.
</ResponseField>

<ResponseField name="fontFamily" field-type="string" required={true}>
  The human-readable font-family name used in CSS (e.g., "Open Sans", "Playfair Display").
</ResponseField>

<ResponseField name="fontFaces" field-type="array" required={true}>
  A list of font-face definitions, specifying variations such as weight and style.

  <Expandable title="Array Items">
    <ResponseField name="weight" field-type="integer" required={true}>
      Numeric representation of the font weight (400=regular, 500=medium, 700=bold, 900=black).
    </ResponseField>

    <ResponseField name="sources" field-type="array" required={true}>
      Font source files provided in supported formats (e.g., woff2, woff).

      <Expandable title="Array Items">
        <ResponseField name="url" field-type="string" required={true} />

        <ResponseField name="format" field-type="string" enum="woff2,woff">
          The format of the font file. Prefer 'woff2' for modern browsers.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="permissions" field-type="object" required={true}>
  The set of permissions for the font definition

  <Expandable title="Object Properties">
    <ResponseField name="edit" field-type="boolean" required={true}>
      Can the user edit the font definition
    </ResponseField>
  </Expandable>
</ResponseField>

`404` Not Found

<ResponseField name="error" field-type="object" required={true}>
  <Expandable title="Object Properties">
    <ResponseField name="code" field-type="integer" required={true} enum="404" />

    <ResponseField name="message" field-type="string" required={true} />
  </Expandable>
</ResponseField>

`412` PreconditionFailed

<ResponseField name="error" field-type="object" required={true}>
  <Expandable title="Object Properties">
    <ResponseField name="code" field-type="integer" required={true} enum="412" />

    <ResponseField name="message" field-type="string" required={true} />
  </Expandable>
</ResponseField>
