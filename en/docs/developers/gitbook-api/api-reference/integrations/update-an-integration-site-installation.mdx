Updates the configuration of an integration for one site.

**PATCH** `/integrations/{integrationName}/installations/{installationId}/sites/{siteId}`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="integrationName" param-type="string" required={true}>
  Name of the integration.
</ParamField>

<ParamField path="installationId" param-type="string" required={true}>
  Identifier of the installation
</ParamField>

<ParamField path="siteId" param-type="string" required={true}>
  The unique id of the site
</ParamField>

<ParamField query="extended" param-type="boolean">
  If true, returns the site object in each items. If false, returns the site ID in each items.
</ParamField>

**Request body**

<ParamField body="externalIds" param-type="string[]">
  External IDs assigned by the integration.
</ParamField>

<ParamField body="configuration" param-type="object">
  Configuration of the integration at the account level
</ParamField>

**Responses**

`200` The site installation has been updated.

<ResponseField name="integration" field-type="string" required={true}>
  Unique name identifier of the integration
</ResponseField>

<ResponseField name="installation" field-type="string" required={true}>
  ID of the integration installation
</ResponseField>

<ResponseField name="status" field-type="string" required={true} enum="active,pending,paused" />

<ResponseField name="configuration" field-type="object" required={true}>
  Configuration of the integration for this site
</ResponseField>

<ResponseField name="externalIds" field-type="string[]" required={true}>
  External IDs assigned by the integration.
</ResponseField>

<ResponseField name="urls" field-type="object" required={true}>
  URLs associated with the object

  <Expandable title="Object Properties">
    <ResponseField name="location" field-type="string" required={true}>
      URL of the integration's installation in the API
    </ResponseField>

    <ResponseField name="publicEndpoint" field-type="string" required={true}>
      Public HTTP endpoint for the integration's installation
    </ResponseField>

    <ResponseField name="publicContentEndpoint" field-type="string" required={true}>
      Public endpoint serving the integration's rendered content (e.g. WebFrames), isolated from the cookie-bearing publicEndpoint origin. Equals publicEndpoint when no separate content origin is configured.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="site" field-type="string" required={true}>
  The site the integration is installed on. Using the string value is deprecated in favor of site.id
</ResponseField>
