**GET** `/integrations/{integrationName}/installations`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="integrationName" param-type="string" required={true}>
  Name of the integration.
</ParamField>

<ParamField query="page" param-type="string">
  Identifier of the page results to fetch.
</ParamField>

<ParamField query="limit" param-type="number">
  The number of results per page
</ParamField>

<ParamField query="externalId" param-type="string">
  External Id to filter by
</ParamField>

**Responses**

`200` OK

<ResponseField name="next" field-type="object">
  <Expandable title="Object Properties">
    <ResponseField name="page" field-type="string" required={true}>
      Unique identifier to query the next results page
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="count" field-type="number">
  Total count of objects in the list
</ResponseField>

<ResponseField name="items" field-type="array" required={true}>
  <Expandable title="Array Items">
    <ResponseField name="id" field-type="string" required={true} />

    <ResponseField name="status" field-type="string" required={true} enum="active,pending,paused" />

    <ResponseField name="space_selection" field-type="string" required={true} enum="all,selected">
      Describe whether all spaces have been selected or there's a selection involved
    </ResponseField>

    <ResponseField name="site_selection" field-type="string" required={true} enum="all,selected">
      Describe whether all sites have been selected or there's a selection involved
    </ResponseField>

    <ResponseField name="spaces" field-type="number" required={true}>
      Count of spaces, the installation is managing
    </ResponseField>

    <ResponseField name="configuration" field-type="object" required={true}>
      Configuration of the integration at the account level
    </ResponseField>

    <ResponseField name="createdAt" field-type="string" required={true} />

    <ResponseField name="updatedAt" field-type="string" required={true} />

    <ResponseField name="urls" field-type="object" required={true}>
      URLs associated with the object

      <Expandable title="Object Properties">
        <ResponseField name="location" field-type="string" required={true}>
          URL of the installation in the API
        </ResponseField>

        <ResponseField name="app" field-type="string" required={true}>
          URL of the integration's installation in the application
        </ResponseField>

        <ResponseField name="publicEndpoint" field-type="string" required={true}>
          Public HTTP endpoint for the integration's installation
        </ResponseField>

        <ResponseField name="publicContentEndpoint" field-type="string" required={true}>
          Public endpoint serving the integration's rendered content (e.g. WebFrames), isolated from the cookie-bearing publicEndpoint origin. Equals publicEndpoint when no separate content origin is configured.
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="externalIds" field-type="string[]" required={true}>
      External IDs assigned by the integration.
    </ResponseField>

    <ResponseField name="target" field-type="string" required={true} />

    <ResponseField name="network" field-type="object">
      Network configuration for the installation

      <Expandable title="Object Properties">
        <ResponseField name="proxy" field-type="boolean">
          Whether the installation is proxied through the backend
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>
