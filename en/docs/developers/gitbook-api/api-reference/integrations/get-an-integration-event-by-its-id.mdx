**GET** `/integrations/{integrationName}/events/{eventId}`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="integrationName" param-type="string" required={true}>
  Name of the integration.
</ParamField>

<ParamField path="eventId" param-type="string" required={true}>
  ID of the integration event
</ParamField>

**Responses**

`200` Integration event

<ResponseField name="event" field-type="object" required={true}>
  <Expandable title="Object Properties">
    <ResponseField name="id" field-type="string" required={true}>
      Unique ID of the event.
    </ResponseField>

    <ResponseField name="integrationId" field-type="string" required={true}>
      Unique ID of the integration.
    </ResponseField>

    <ResponseField name="installationId" field-type="string">
      Unique ID of the integration installation.
    </ResponseField>

    <ResponseField name="createdAt" field-type="string" required={true} />

    <ResponseField name="payload" field-type="string" required={true}>
      Any event that can be received from GitBook.
    </ResponseField>

    <ResponseField name="status" field-type="string" required={true} enum="success,failed">
      Status of the event.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="trace" field-type="object">
  <Expandable title="Object Properties">
    <ResponseField name="logs" field-type="array" required={true}>
      <Expandable title="Array Items">
        <ResponseField name="message" field-type="string">
          The message of the log entry.
        </ResponseField>

        <ResponseField name="timestamp" field-type="string" />

        <ResponseField name="level" field-type="string" enum="debug,info,warn,error">
          The level of the log entry.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

`404` Not Found

<ResponseField name="error" field-type="object" required={true}>
  <Expandable title="Object Properties">
    <ResponseField name="code" field-type="integer" required={true} enum="404" />

    <ResponseField name="message" field-type="string" required={true} />
  </Expandable>
</ResponseField>
