Retrieves metadata for a single file (image or attachment) stored in the change request's current revision. Use this to get the download URL or details of a specific asset referenced by a page in the change request.

**GET** `/spaces/{spaceId}/change-requests/{changeRequestId}/content/files/{fileId}`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="spaceId" param-type="string" required={true}>
  The unique id of the space
</ParamField>

<ParamField path="changeRequestId" param-type="string" required={true}>
  The unique ID of the change request or its number identifier in the space
</ParamField>

<ParamField path="fileId" param-type="string" required={true}>
  The unique id of the file
</ParamField>

<ParamField query="metadata" param-type="boolean">
  If `false` is passed, "git" mutable metadata will not returned. Passing `false` can optimize performances of the lookup.
</ParamField>

<ParamField query="computed" param-type="boolean">
  If `false` is passed, content will not be computed
</ParamField>

**Responses**

`200` OK

<ResponseField name="id" field-type="string" required={true} />

<ResponseField name="name" field-type="string" required={true} />

<ResponseField name="contentType" field-type="string" required={true} />

<ResponseField name="downloadURL" field-type="string" required={true} />

<ResponseField name="size" field-type="number" required={true} />

<ResponseField name="dimensions" field-type="object">
  For images, it contains the dimensions of it.

  <Expandable title="Object Properties">
    <ResponseField name="width" field-type="number" required={true} />

    <ResponseField name="height" field-type="number" required={true} />
  </Expandable>
</ResponseField>

<ResponseField name="git" field-type="object">
  <Expandable title="Object Properties">
    <ResponseField name="oid" field-type="string" required={true}>
      SHA for the blob
    </ResponseField>

    <ResponseField name="path" field-type="string" required={true}>
      Path of the blob in the Git tree
    </ResponseField>
  </Expandable>
</ResponseField>
