This API powers the inline discussion around any new or updated documentation. Participate in comment threads and resolve them after reaching consensus.

## The Comment object

<ResponseField name="object" field-type="string" required={true} enum="comment">
  Type of Object, always equals to "comment"
</ResponseField>

<ResponseField name="id" field-type="string" required={true}>
  Unique identifier for the comment.
</ResponseField>

<ResponseField name="postedBy" field-type="object" required={true}>
  <Expandable title="Object Properties">
    <ResponseField name="object" field-type="string" required={true} enum="user">
      Type of Object, always equals to "user"
    </ResponseField>

    <ResponseField name="id" field-type="string" required={true}>
      Unique identifier for the user
    </ResponseField>

    <ResponseField name="displayName" field-type="string" required={true}>
      Full name for the user
    </ResponseField>

    <ResponseField name="email" field-type="string">
      Email address of the user
    </ResponseField>

    <ResponseField name="photoURL" field-type="string">
      URL of the user's profile picture
    </ResponseField>

    <ResponseField name="urls" field-type="object" required={true}>
      URLs associated with the object

      <Expandable title="Object Properties">
        <ResponseField name="location" field-type="string" required={true}>
          URL of the user in the API
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="postedAt" field-type="string" required={true} />

<ResponseField name="editedAt" field-type="string" />

<ResponseField name="reactions" field-type="array" required={true}>
  <Expandable title="Array Items">
    <ResponseField name="emoji" field-type="string" required={true}>
      The Emoji of the reaction
    </ResponseField>

    <ResponseField name="count" field-type="number" required={true}>
      The number of users who reacted with this emoji
    </ResponseField>

    <ResponseField name="users" field-type="array" required={true}>
      The users who reacted with this emoji

      <Expandable title="Array Items">
        <ResponseField name="user" field-type="object" required={true}>
          <Expandable title="Object Properties">
            <ResponseField name="object" field-type="string" required={true} enum="user">
              Type of Object, always equals to "user"
            </ResponseField>

            <ResponseField name="id" field-type="string" required={true}>
              Unique identifier for the user
            </ResponseField>

            <ResponseField name="displayName" field-type="string" required={true}>
              Full name for the user
            </ResponseField>

            <ResponseField name="email" field-type="string">
              Email address of the user
            </ResponseField>

            <ResponseField name="photoURL" field-type="string">
              URL of the user's profile picture
            </ResponseField>

            <ResponseField name="urls" field-type="object" required={true}>
              URLs associated with the object

              <Expandable title="Object Properties">
                <ResponseField name="location" field-type="string" required={true}>
                  URL of the user in the API
                </ResponseField>
              </Expandable>
            </ResponseField>
          </Expandable>
        </ResponseField>

        <ResponseField name="reactedAt" field-type="string" required={true} />
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="replies" field-type="number" required={true}>
  The number of replies to this comment.
</ResponseField>

<ResponseField name="repliers" field-type="array" required={true}>
  The users who replied to this comment.

  <Expandable title="Array Items">
    <ResponseField name="updatedAt" field-type="string" required={true} />

    <ResponseField name="count" field-type="integer" required={true} />

    <ResponseField name="user" field-type="object" required={true}>
      <Expandable title="Object Properties">
        <ResponseField name="object" field-type="string" required={true} enum="user">
          Type of Object, always equals to "user"
        </ResponseField>

        <ResponseField name="id" field-type="string" required={true}>
          Unique identifier for the user
        </ResponseField>

        <ResponseField name="displayName" field-type="string" required={true}>
          Full name for the user
        </ResponseField>

        <ResponseField name="email" field-type="string">
          Email address of the user
        </ResponseField>

        <ResponseField name="photoURL" field-type="string">

        </ResponseField>

        <ResponseField name="urls" field-type="object" required={true}>

          <Expandable title="Object Properties">
            <ResponseField name="location" field-type="string" required={true}>
              URL of the user in the API
            </ResponseField>
          </Expandable>
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="body" field-type="string" required={true} />

<ResponseField name="target" field-type="object" required={true}>
  Information about the target of the comment.

  <Expandable title="Object Properties">
    <ResponseField name="node" field-type="object">
      The node this comment is attached to.

      <Expandable title="Object Properties">
        <ResponseField name="id" field-type="string" required={true} />
      </Expandable>
    </ResponseField>

    <ResponseField name="tableCell" field-type="object">
      Details about the table cell this comment is attached to, if any.

      <Expandable title="Object Properties">
        <ResponseField name="record" field-type="string" required={true} />

        <ResponseField name="definition" field-type="string" required={true} />
      </Expandable>
    </ResponseField>

    <ResponseField name="changeRequest" field-type="string">
      The change request containing this comment, if the comment was made inside a change request.
    </ResponseField>

    <ResponseField name="review" field-type="string">
      The review containing this comment, if this comment was made as part of a review.
    </ResponseField>

    <ResponseField name="page" field-type="object">
      Information about the page, if this comment refers to a specific page.

      <Expandable title="Object Properties">
        <ResponseField name="id" field-type="string" required={true}>
          The ID of the page
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="space" field-type="string" required={true}>
      The space containing this comment.
    </ResponseField>

    <ResponseField name="revision" field-type="string" required={true}>
      The revision in which the target can be found in.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="urls" field-type="object" required={true}>
  URLs associated with the object

  <Expandable title="Object Properties">
    <ResponseField name="location" field-type="string" required={true}>
      URL of the comment in the API
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="permissions" field-type="object" required={true}>
  The set of permissions for the comment

  <Expandable title="Object Properties">
    <ResponseField name="view" field-type="boolean" required={true}>
      Can the user view the comment.
    </ResponseField>

    <ResponseField name="edit" field-type="boolean" required={true}>
      Can the user edit the comment.
    </ResponseField>

    <ResponseField name="reply" field-type="boolean">
      Can the user react or send a reply to the comment.
    </ResponseField>

    <ResponseField name="delete" field-type="boolean" required={true}>
      Can the user delete the comment.
    </ResponseField>
  </Expandable>
</ResponseField>
