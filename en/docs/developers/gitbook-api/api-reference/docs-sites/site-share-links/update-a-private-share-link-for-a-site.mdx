Updates a share link's properties. Use this to temporarily disable a share link without losing its configuration.

**PATCH** `/orgs/{organizationId}/sites/{siteId}/share-links/{shareLinkId}`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="organizationId" param-type="string" required={true}>
  The unique id of the organization
</ParamField>

<ParamField path="siteId" param-type="string" required={true}>
  The unique id of the site
</ParamField>

<ParamField path="shareLinkId" param-type="string" required={true}>
  The unique id of the share link
</ParamField>

**Request body**

<ParamField body="active" param-type="boolean" />

<ParamField body="name" param-type="string">
  Name of the share link
</ParamField>

**Responses**

`200` The site share link has been updated

<ResponseField name="object" field-type="string" required={true} enum="share-link">
  Type of Object, always equals to "share-link"
</ResponseField>

<ResponseField name="id" field-type="string" required={true}>
  Unique identifier for the share-link
</ResponseField>

<ResponseField name="createdAt" field-type="string" required={true} />

<ResponseField name="name" field-type="string">
  Name of the share link
</ResponseField>

<ResponseField name="active" field-type="boolean" />

<ResponseField name="urls" field-type="object" required={true}>
  URLs associated with the object

  <Expandable title="Object Properties">
    <ResponseField name="published" field-type="string">
      URL of the published version of the share-link.
    </ResponseField>
  </Expandable>
</ResponseField>
