Configures a new channel integration that allows GitBook Agent to operate within an external tool connected to this site. The channel role determines whether the Agent acts as a support assistant — answering questions sourced from the site's documentation — or as a collaborator that can open change requests and sync documentation updates directly from the connected platform.

**POST** `/orgs/{organizationId}/sites/{siteId}/channels`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="organizationId" param-type="string" required={true}>
  The unique id of the organization
</ParamField>

<ParamField path="siteId" param-type="string" required={true}>
  The unique id of the site
</ParamField>

**Request body**

<ParamField body="role" param-type="string" enum="collaborator,support" />

**Responses**

`201` Site channel created

<ResponseField name="object" field-type="string" required={true} enum="site-channel" />

<ResponseField name="id" field-type="string" required={true}>
  Unique identifier for the site channel.
</ResponseField>

<ResponseField name="status" field-type="string" required={true} enum="active,pending" />

<ResponseField name="role" field-type="string" required={true} enum="collaborator,support" />

<ResponseField name="answers" field-type="integer" required={true}>
  Number of answers linked to this channel.
</ResponseField>

<ResponseField name="createdAt" field-type="string" required={true} />

<ResponseField name="urls" field-type="object" required={true}>

    </ResponseField>
  </Expandable>
</ResponseField>
