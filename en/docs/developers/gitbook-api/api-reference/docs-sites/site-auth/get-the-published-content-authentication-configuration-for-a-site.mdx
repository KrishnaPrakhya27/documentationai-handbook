Retrieves the authentication settings that protect a site's published content. Use this to review the current auth method before making changes or troubleshooting access issues.

**GET** `/orgs/{organizationId}/sites/{siteId}/publishing/auth`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="organizationId" param-type="string" required={true}>
  The unique id of the organization
</ParamField>

<ParamField path="siteId" param-type="string" required={true}>
  The unique id of the site
</ParamField>

**Responses**

`200` OK

<ResponseField name="object" field-type="string" required={true} enum="publishing-auth">
  Type of Object, always equals to "publishing-auth"
</ResponseField>

<ResponseField name="privateKey" field-type="string" required={true}>
  Private key used to sign JWT tokens.
</ResponseField>

<ResponseField name="fallbackURL" field-type="string">
  URL to redirect to when the authenticated access secret is invalid.
</ResponseField>

<ResponseField name="logoutURL" field-type="string">
  URL used to log visitors out from the upstream custom auth backend.
</ResponseField>

<ResponseField name="integration" field-type="string">
  Name of the authenticated access integration installed on the site (if any).
  It is also the one being used as VA backend when the published auth settings are configured to use "integration" as backend.
</ResponseField>

`400` Bad Request

<ResponseField name="error" field-type="object" required={true}>
  <Expandable title="Object Properties">
    <ResponseField name="code" field-type="integer" required={true} enum="400" />

    <ResponseField name="message" field-type="string" required={true} />
  </Expandable>
</ResponseField>
