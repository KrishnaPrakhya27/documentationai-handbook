External links add absolute HTTP or HTTPS destinations alongside sections and groups in a site's navigation.

## The SiteExternalLink object

<ResponseField name="object" field-type="string" required={true} enum="site-external-link">
  The object type, which is always "site-external-link"
</ResponseField>

<ResponseField name="id" field-type="string" required={true}>
  Unique identifier of the external link
</ResponseField>

<ResponseField name="title" field-type="string" required={true}>
  Title of the site section
</ResponseField>

<ResponseField name="localizedTitle" field-type="object">
  Localized string value with a 128 character limit, keyed by locale. Contains overrides for non-default languages only.
</ResponseField>

<ResponseField name="description" field-type="string">
  Description of the site section
</ResponseField>

<ResponseField name="localizedDescription" field-type="object">
  Localized string value with a 256 character limit, keyed by locale. Contains overrides for non-default languages only.
</ResponseField>

<ResponseField name="icon" field-type="string">
  Name of the icon
</ResponseField>

<ResponseField name="url" field-type="string" required={true} />

<ResponseField name="sectionGroup" field-type="string">
  ID of the section group the external link belongs to. Omitted for a root link.
</ResponseField>

<ResponseField name="draft" field-type="boolean" required={true}>
  Whether the external link is draft and not live
</ResponseField>

<ResponseField name="condition" field-type="string">
  Expression to evaluate
</ResponseField>
