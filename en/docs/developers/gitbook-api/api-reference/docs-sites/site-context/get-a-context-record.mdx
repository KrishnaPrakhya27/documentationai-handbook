**GET** `/orgs/{organizationId}/sites/{siteId}/context-records/{siteContextRecordId}`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="organizationId" param-type="string" required={true}>
  The unique id of the organization
</ParamField>

<ParamField path="siteId" param-type="string" required={true}>
  The unique id of the site
</ParamField>

<ParamField path="siteContextRecordId" param-type="string" required={true}>
  The unique id of the context record
</ParamField>

**Responses**

`200` OK

<ResponseField name="object" field-type="string" required={true} enum="context-record" />

<ResponseField name="id" field-type="string" required={true}>
  The ID of the record
</ResponseField>

<ResponseField name="title" field-type="string" required={true}>
  Title for the context record
</ResponseField>

<ResponseField name="description" field-type="string">
  Optional short description of the context record
</ResponseField>

<ResponseField name="type" field-type="string" required={true} enum="ticket,document" />

<ResponseField name="createdAt" field-type="string" required={true} />

<ResponseField name="processedAt" field-type="string" />

<ResponseField name="connection" field-type="string" required={true}>
  The connection ID associated with the record
</ResponseField>

<ResponseField name="connector" field-type="string" required={true} />

<ResponseField name="topics" field-type="string[]">
  The associated topics of the record
</ResponseField>

<ResponseField name="urls" field-type="object" required={true}>
  URLs associated with the record

  <Expandable title="Object Properties">
    <ResponseField name="location" field-type="string" required={true}>
      URL of the record in the API
    </ResponseField>

    <ResponseField name="source" field-type="string">
      URL of the record in the source
    </ResponseField>
  </Expandable>
</ResponseField>
