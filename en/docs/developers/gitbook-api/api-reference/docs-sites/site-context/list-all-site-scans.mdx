**GET** `/orgs/{organizationId}/sites/{siteId}/scans`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="organizationId" param-type="string" required={true}>
  The unique id of the organization
</ParamField>

<ParamField path="siteId" param-type="string" required={true}>
  The unique id of the site
</ParamField>

<ParamField query="page" param-type="string">
  Identifier of the page results to fetch.
</ParamField>

<ParamField query="limit" param-type="number">
  The number of results per page
</ParamField>

<ParamField query="topic" param-type="string">
  Filter scans by associated site topic ID.
</ParamField>

<ParamField query="status" param-type="string" enum="pending,running,done,failed">
  Filter scans by status.
</ParamField>

**Responses**

`200` OK

<ResponseField name="next" field-type="object">
  <Expandable title="Object Properties">
    <ResponseField name="page" field-type="string" required={true}>
      Unique identifier to query the next results page
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="count" field-type="number">
  Total count of objects in the list
</ResponseField>

<ResponseField name="items" field-type="array" required={true}>
  <Expandable title="Array Items">
    <ResponseField name="object" field-type="string" required={true} enum="site-scan" />

    <ResponseField name="id" field-type="string" required={true}>
      The ID of the scan
    </ResponseField>

    <ResponseField name="status" field-type="string" required={true} enum="pending,running,done,failed">
      Status of a persisted site scan.
    </ResponseField>

    <ResponseField name="topic" field-type="string" required={true}>
      The site topic ID associated with the scan. It is null for scans that are not associated with a current topic in the site.
    </ResponseField>

    <ResponseField name="createdAt" field-type="string" required={true} />

    <ResponseField name="startedAt" field-type="string" />

    <ResponseField name="finishedAt" field-type="string" />

    <ResponseField name="findings" field-type="object" required={true}>
      Number of findings created by the scan, grouped by severity.

      <Expandable title="Object Properties">
        <ResponseField name="low" field-type="number" required={true} />

        <ResponseField name="medium" field-type="number" required={true} />

        <ResponseField name="high" field-type="number" required={true} />
      </Expandable>
    </ResponseField>

    <ResponseField name="failureMessage" field-type="string">
      Failure message for failed scans
    </ResponseField>

    <ResponseField name="urls" field-type="object" required={true}>
      URLs associated with the scan

      <Expandable title="Object Properties">
        <ResponseField name="location" field-type="string" required={true}>
          URL of the scan endpoint in the API
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>
