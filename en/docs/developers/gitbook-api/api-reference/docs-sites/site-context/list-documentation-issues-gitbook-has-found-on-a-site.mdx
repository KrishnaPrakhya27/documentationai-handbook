Lists the quality issues GitBook's AI has identified on a site by comparing its content against real usage and external sources — content gaps, outdated or incorrect content, and broken or outbound links. Use this when asked to find, audit, or fix problems in a site's docs.

**GET** `/orgs/{organizationId}/sites/{siteId}/findings`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="organizationId" param-type="string" required={true}>
  The unique id of the organization
</ParamField>

<ParamField path="siteId" param-type="string" required={true}>
  The unique id of the site
</ParamField>

<ParamField query="page" param-type="string">
  Identifier of the page results to fetch.
</ParamField>

<ParamField query="limit" param-type="number">
  The number of results per page
</ParamField>

<ParamField query="topic" param-type="string">
  Filter findings by associated site topic ID.
</ParamField>

<ParamField query="question" param-type="string">
  Filter findings that reference the given site question ID as evidence.
</ParamField>

<ParamField query="status" param-type="string" enum="candidate,open,done,canceled">
  Filter findings by status.
</ParamField>

<ParamField query="type" param-type="string[]" enum="content-outdated,content-incorrect,content-gap,broken-link,outbound-link-error,other">
  Filter findings by type. When multiple types are given, findings matching any of them are returned.
</ParamField>

<ParamField query="severity" param-type="string" enum="low,medium,high">
  Filter findings by estimated severity.
</ParamField>

<ParamField query="hasChangeRequests" param-type="boolean">
  Filter findings that have at least one associated change request.
</ParamField>

<ParamField query="source" param-type="string[]">
  Filter findings by the source of their evidence. Each entry is either `connection:<connectionId>` (findings with at least one piece of evidence coming from the given connection) or `questions` (findings with at least one piece of evidence coming from a site question). When multiple entries are given, findings matching any of them are returned.
</ParamField>

<ParamField query="updatedAfter" param-type="string">
  Only return findings last updated at or after this date.
</ParamField>

<ParamField query="updatedBefore" param-type="string">
  Only return findings last updated before this date.
</ParamField>

<ParamField query="orderBy" param-type="string" enum="updatedAt,severity,sources">
  Sort findings by the given field.
</ParamField>

    <ResponseField name="status" field-type="string" required={true} enum="candidate,open,done,canceled">
      Status of a site finding.

      - `candidate` is a finding that has been raised but not yet confirmed. Typically these are not shown to users.
      - `open` is a finding that has been surfaced to users and is still under investigation.
      - `done` is a finding that has been resolved and is no longer under investigation.
      - `canceled` is a finding that has been canceled and is no longer under investigation.
    </ResponseField>

    <ResponseField name="type" field-type="string" required={true} enum="content-outdated,content-incorrect,content-gap,broken-link,outbound-link-error,other">
      Classification of a site finding:

      - Outdated content: The content was correct but is no longer up to date with external context and may be misleading or incorrect.
      - Incorrect content: The content is factually wrong or misleading, leading users astray regardless of whether it was ever correct.
      - Content gap: The content is missing important information that should be present to answer user questions effectively.
      - Broken link: A URL on the site returns a 404, and can be resolved by creating a redirect or a new page.
      - Outbound link error: An external URL used by the site's content fails to resolve, and can be fixed by replacing or removing the link.
      - Other: Any other type of finding not covered by the above categories.
    </ResponseField>

    <ResponseField name="severity" field-type="string" required={true} enum="low,medium,high">
      Estimated end-user severity of a site finding.
    </ResponseField>
