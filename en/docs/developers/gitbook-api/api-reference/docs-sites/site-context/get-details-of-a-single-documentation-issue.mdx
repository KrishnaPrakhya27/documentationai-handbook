Retrieves the full detail of one documentation issue found by GitBook's AI. Use this after `listSiteFindings` to inspect a specific issue before deciding to fix, dismiss, or investigate it further.

**GET** `/orgs/{organizationId}/sites/{siteId}/findings/{siteFindingId}`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="organizationId" param-type="string" required={true}>
  The unique id of the organization
</ParamField>

<ParamField path="siteId" param-type="string" required={true}>
  The unique id of the site
</ParamField>

<ParamField path="siteFindingId" param-type="string" required={true}>
  The unique id of the site finding
</ParamField>

**Responses**

`200` OK

<ResponseField name="object" field-type="string" required={true} enum="site-finding" />

<ResponseField name="id" field-type="string" required={true}>
  The ID of the finding
</ResponseField>

<ResponseField name="scan" field-type="string" required={true}>
  The site scan ID associated with the finding
</ResponseField>

<ResponseField name="topic" field-type="string" required={true}>
  The site topic ID associated with the finding. It is null for findings that are not associated with a current topic in the site.
</ResponseField>

<ResponseField name="title" field-type="string" required={true}>
  Short finding title
</ResponseField>

<ResponseField name="changeRequests" field-type="number" required={true}>
  Number of change requests linked to the finding
</ResponseField>

<ResponseField name="pages" field-type="number" required={true}>
  Number of pages linked to the finding
</ResponseField>

<ResponseField name="records" field-type="number" required={true}>
  Number of records linked to the finding
</ResponseField>

<ResponseField name="questions" field-type="number" required={true}>
  Number of questions linked to the finding
</ResponseField>

<ResponseField name="status" field-type="string" required={true} enum="candidate,open,done,canceled">
  Status of a site finding.

  - `candidate` is a finding that has been raised but not yet confirmed. Typically these are not shown to users.
  - `open` is a finding that has been surfaced to users and is still under investigation.
  - `done` is a finding that has been resolved and is no longer under investigation.
  - `canceled` is a finding that has been canceled and is no longer under investigation.
</ResponseField>

<ResponseField name="type" field-type="string" required={true} enum="content-outdated,content-incorrect,content-gap,broken-link,outbound-link-error,other">
  Classification of a site finding:

  - Outdated content: The content was correct but is no longer up to date with external context and may be misleading or incorrect.
  - Incorrect content: The content is factually wrong or misleading, leading users astray regardless of whether it was ever correct.
  - Content gap: The content is missing important information that should be present to answer user questions effectively.
  - Broken link: A URL on the site returns a 404, and can be resolved by creating a redirect or a new page.
  - Outbound link error: An external URL used by the site's content fails to resolve, and can be fixed by replacing or removing the link.
  - Other: Any other type of finding not covered by the above categories.
</ResponseField>

<ResponseField name="severity" field-type="string" required={true} enum="low,medium,high">
  Estimated end-user severity of a site finding.
</ResponseField>

<ResponseField name="description" field-type="object" required={true}>
  <Expandable title="Object Properties">
    <ResponseField name="object" field-type="string" required={true} enum="document" />

    <ResponseField name="data" field-type="object" required={true}>
      <Expandable title="Object Properties">
        <ResponseField name="schemaVersion" field-type="integer">
          The schema version of the document. If undefined, the document is considered to be of the latest schema version.
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="nodes" field-type="string[]" required={true} />

    <ResponseField name="meta" field-type="object">
      <Expandable title="Object Properties">
        <ResponseField name="token" field-type="string">
          A content token that can be used to fetch the content referenced in this document from the API.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="autoFixable" field-type="boolean" required={true}>
  Whether an AI agent should be able to fix this finding directly
</ResponseField>

<ResponseField name="brokenLinkAction" field-type="object">
  Suggested fix for a broken-link finding, shaped as the input for `createSiteRedirect` so it can be applied directly. Absent when the finding has no directly applicable redirect (for example, a suggestion to create a new page).

  <Expandable title="Object Properties">
    <ResponseField name="source" field-type="string" required={true}>
      The source path to redirect from.
    </ResponseField>

    <ResponseField name="captureWildcard" field-type="boolean">
      Whether the redirect appends the wildcard-matched suffix of the source path to the destination.
    </ResponseField>

    <ResponseField name="destination" field-type="object" required={true}>
      <Expandable title="Object Properties">
        <ResponseField name="kind" field-type="string" required={true} enum="path" />

        <ResponseField name="spaceId" field-type="string" required={true}>
          Unique identifier for the space containing the page referenced by the path.
        </ResponseField>

        <ResponseField name="path" field-type="string" required={true}>
          Path to the page within the space revision to resolve.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="outboundLink" field-type="object">
  Details of an `outbound-link-error` finding: the failing external URL and the observed failure, sourced from the site's outbound link checks. Absent on findings of any other type.

  <Expandable title="Object Properties">
    <ResponseField name="url" field-type="string" required={true}>
      The external URL that fails to resolve.
    </ResponseField>

    <ResponseField name="status" field-type="string" required={true} enum="pending,healthy,suspect,failed,recovering,canceled">
      Current health status of the URL for this site.
    </ResponseField>

    <ResponseField name="failureKind" field-type="string" enum="client-error,server-error,redirect,dns,tls,connection,timeout">
      Classification of the confirmed failure.
    </ResponseField>

    <ResponseField name="httpStatus" field-type="number">
      HTTP status code of the last observation, when one was received.
    </ResponseField>

    <ResponseField name="checkedAt" field-type="string" />
  </Expandable>
</ResponseField>

<ResponseField name="createdAt" field-type="string" required={true} />

<ResponseField name="updatedAt" field-type="string" required={true} />

<ResponseField name="urls" field-type="object" required={true}>
  URLs associated with the finding

  <Expandable title="Object Properties">
    <ResponseField name="location" field-type="string" required={true}>
      URL of the finding list endpoint in the API
    </ResponseField>
  </Expandable>
</ResponseField>
