**PATCH** `/orgs/{organizationId}/sites/{siteId}/topics/{siteTopicId}`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="organizationId" param-type="string" required={true}>
  The unique id of the organization
</ParamField>

<ParamField path="siteId" param-type="string" required={true}>
  The unique id of the site
</ParamField>

<ParamField path="siteTopicId" param-type="string" required={true}>
  The unique id of the topic
</ParamField>

**Request body**

<ParamField body="usageSettings" param-type="object" required={true}>
  <Expandable title="Object Properties">
    <ParamField body="changeRequests" param-type="boolean" required={true}>
      Whether to use context to propose change requests
    </ParamField>
  </Expandable>
</ParamField>

`200` OK
