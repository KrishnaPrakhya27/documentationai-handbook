Returns a paginated list of redirect rules for the site. Supports filtering by draft status and searching by path. Use this to audit redirects, find a specific rule before updating it, or check whether draft redirects are pending publication.

**GET** `/orgs/{organizationId}/sites/{siteId}/redirects`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="organizationId" param-type="string" required={true}>
  The unique id of the organization
</ParamField>

<ParamField path="siteId" param-type="string" required={true}>
  The unique id of the site
</ParamField>

<ParamField query="page" param-type="string">
  Identifier of the page results to fetch.
</ParamField>

<ParamField query="limit" param-type="number">
  The number of results per page
</ParamField>

<ParamField query="search" param-type="string">
  Search for a redirect by path
</ParamField>

<ParamField query="draft" param-type="boolean">
  Filter redirects by draft mode.
</ParamField>

<ParamField query="orderBy" param-type="string" enum="createdAt,updatedAt,source">
  Sort redirects by the given field.
</ParamField>

**Responses**

`200` The list of redirects for the site.

<ResponseField name="next" field-type="object">
  <Expandable title="Object Properties">
    <ResponseField name="page" field-type="string" required={true}>
      Unique identifier to query the next results page
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="count" field-type="number">
  Total count of objects in the list
</ResponseField>

<ResponseField name="items" field-type="array" required={true}>
  <Expandable title="Array Items">
    <ResponseField name="object" field-type="string" required={true} enum="site-redirect" />

    <ResponseField name="id" field-type="string" required={true}>
      The unique identifier for the redirect.
    </ResponseField>

    <ResponseField name="source" field-type="string" required={true}>
      The source path to redirect from.
    </ResponseField>

    <ResponseField name="draft" field-type="boolean" required={true}>
      Whether the redirect is draft and not live.
    </ResponseField>

    <ResponseField name="captureWildcard" field-type="boolean" required={true}>
      Capture and append the wildcard-matched suffix to the destination URL.
    </ResponseField>

    <ResponseField name="permanent" field-type="boolean" required={true}>
      Whether the redirect is configured as permanent. Permanent redirects are served as HTTP 308 for public live traffic and HTTP 307 in non-public contexts.
    </ResponseField>

    <ResponseField name="destination" field-type="string" required={true} />

    <ResponseField name="createdAt" field-type="string" required={true}>
      When the redirect was created.
    </ResponseField>

    <ResponseField name="updatedAt" field-type="string">
      When the redirect was last updated. Absent if never updated since creation.
    </ResponseField>
  </Expandable>
</ResponseField>
