Inspect the canonical questions asked on your site, the generated answers linked to each question, and the sources used for each answer.

## The SiteQuestion object

<ResponseField name="object" field-type="string" required={true} enum="site-question" />

<ResponseField name="id" field-type="string" required={true}>
  The ID of the site question.
</ResponseField>

<ResponseField name="question" field-type="string" required={true}>
  Canonical text for the unique site question.
</ResponseField>

<ResponseField name="type" field-type="string" required={true} enum="exploring,how-to,troubleshooting,reference,unknown">
  Type classification of the user question.
</ResponseField>

<ResponseField name="createdAt" field-type="string" required={true} />

<ResponseField name="askedAt" field-type="string" required={true} />

<ResponseField name="stats" field-type="object" required={true}>
  Aggregated stats for questions and linked answers.

  <Expandable title="Object Properties">
    <ResponseField name="total" field-type="number" required={true}>
      Total count of questions included in the aggregation.
    </ResponseField>

    <ResponseField name="answers" field-type="number" required={true}>
      Total count of answers linked to the questions.
    </ResponseField>

    <ResponseField name="feedback" field-type="object" required={true}>
      Aggregated feedback for answers linked to a site question.

      <Expandable title="Object Properties">
        <ResponseField name="positive" field-type="number" required={true}>
          Count of answers with positive feedback.
        </ResponseField>

        <ResponseField name="negative" field-type="number" required={true}>
          Count of answers with negative feedback.
        </ResponseField>

        <ResponseField name="unrated" field-type="number" required={true}>
          Count of answers without feedback.
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="answered" field-type="object" required={true}>
      Aggregated answered assessment for answers linked to a site question.

      <Expandable title="Object Properties">
        <ResponseField name="yes" field-type="number" required={true}>
          Count of answers assessed as fully answered.
        </ResponseField>

        <ResponseField name="partially" field-type="number" required={true}>
          Count of answers assessed as partially answered.
        </ResponseField>

        <ResponseField name="no" field-type="number" required={true}>
          Count of answers assessed as not answered.
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="relevance" field-type="object" required={true}>
      <Expandable title="Object Properties">
        <ResponseField name="relevant" field-type="number" required={true} />

        <ResponseField name="spam" field-type="number" required={true} />

        <ResponseField name="off-topic" field-type="number" required={true} />

        <ResponseField name="unclear" field-type="number" required={true} />
      </Expandable>
    </ResponseField>

    <ResponseField name="types" field-type="object" required={true}>
      <Expandable title="Object Properties">
        <ResponseField name="exploring" field-type="number" required={true} />

        <ResponseField name="how-to" field-type="number" required={true} />

        <ResponseField name="troubleshooting" field-type="number" required={true} />

        <ResponseField name="reference" field-type="number" required={true} />

        <ResponseField name="unknown" field-type="number" required={true} />
      </Expandable>
    </ResponseField>

    <ResponseField name="channelTypes" field-type="object" required={true}>
      Count of answers by site channel type, including core channels.
    </ResponseField>

    <ResponseField name="channels" field-type="object" required={true}>
      Count of answers by site channel ID.
    </ResponseField>

    <ResponseField name="weeklyPulse" field-type="number[]" required={true}>
      Weekly count of related question answers for the past 3 months.
    </ResponseField>
  </Expandable>
</ResponseField>

</ResponseField>
