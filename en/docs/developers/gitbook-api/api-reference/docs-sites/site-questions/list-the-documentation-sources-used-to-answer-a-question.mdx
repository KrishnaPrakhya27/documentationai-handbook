Returns the documentation pages the Ask AI assistant used to answer a given question. Use it to see which content grounded the AI's responses to that question.

**GET** `/orgs/{organizationId}/sites/{siteId}/questions/{siteQuestionId}/sources`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="organizationId" param-type="string" required={true}>
  The unique id of the organization
</ParamField>

<ParamField path="siteId" param-type="string" required={true}>
  The unique id of the site
</ParamField>

<ParamField path="siteQuestionId" param-type="string" required={true}>
  The unique id of the site question
</ParamField>

<ParamField query="page" param-type="string">
  Identifier of the page results to fetch.
</ParamField>

<ParamField query="limit" param-type="number">
  The number of results per page
</ParamField>

<ParamField query="from" param-type="string">
  Filter sources to answers created at or after this timestamp.
</ParamField>

<ParamField query="to" param-type="string">
  Filter sources to answers created at or before this timestamp.
</ParamField>

<ParamField query="relevance" param-type="string[]" enum="relevant,spam,off-topic,unclear">
  Filter answer sources by question relevance.
</ParamField>

<ParamField query="channelType" param-type="string[]">
  Filter answer sources by answer channel type.
</ParamField>

<ParamField query="answered" param-type="string[]" enum="no,partially,yes">
  Filter answer sources by answer resolution.
</ParamField>

`200` OK
