Returns a paginated list of the questions visitors asked a site's Ask AI assistant, each with usage stats. Use it to discover what users are looking for and where the documentation has gaps.

**GET** `/orgs/{organizationId}/sites/{siteId}/questions`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="organizationId" param-type="string" required={true}>
  The unique id of the organization
</ParamField>

<ParamField path="siteId" param-type="string" required={true}>
  The unique id of the site
</ParamField>

<ParamField query="from" param-type="string">
  Filter stats to answers created at or after this timestamp.
</ParamField>

<ParamField query="to" param-type="string">
  Filter stats to answers created at or before this timestamp.
</ParamField>

<ParamField query="type" param-type="string" enum="exploring,how-to,troubleshooting,reference,unknown">
  Filter questions by question type.
</ParamField>

<ParamField query="relevance" param-type="string[]" enum="relevant,spam,off-topic,unclear">
  Filter questions by question relevance.
</ParamField>

<ParamField query="channelType" param-type="string[]">
  Filter questions by answer channel type.
</ParamField>

<ParamField query="answered" param-type="string[]" enum="no,partially,yes">
  Filter questions by answer resolution.
</ParamField>

<ParamField query="topic" param-type="string[]">
  Filter questions by associated site topic IDs.
</ParamField>

<ParamField query="order" param-type="string" enum="asc,desc">
  An order for the items in the list
</ParamField>

<ParamField query="sort" param-type="string" enum="askedAt,stats.answered,stats.feedback,stats.answers">
  Sort questions by latest ask activity, answered rate, or positive feedback rate. If omitted, ordering stays on canonical question creation date to preserve current behavior.
</ParamField>

<ParamField query="page" param-type="string">
  Identifier of the page results to fetch.
</ParamField>

<ParamField query="limit" param-type="number">
  The number of results per page
</ParamField>
