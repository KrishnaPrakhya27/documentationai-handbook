Returns a paginated list of individual Ask AI answers generated for a site. Each is the assistant's response to a specific asked question, with the raw question text, channel, and feedback. Use it to review answer quality and what users asked across the site.

**GET** `/orgs/{organizationId}/sites/{siteId}/answers`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="organizationId" param-type="string" required={true}>
  The unique id of the organization
</ParamField>

<ParamField path="siteId" param-type="string" required={true}>
  The unique id of the site
</ParamField>

<ParamField query="question" param-type="string">
  Filter answers to a specific site question ID.
</ParamField>

<ParamField query="from" param-type="string">
  Filter answers created at or after this timestamp.
</ParamField>

<ParamField query="to" param-type="string">
  Filter answers created at or before this timestamp.
</ParamField>

<ParamField query="language" param-type="string">
  Filter answers by ISO language code.
</ParamField>

<ParamField query="answered" param-type="string[]" enum="no,partially,yes">
  Filter answers by answered resolution.
</ParamField>

<ParamField query="helpfulness" param-type="string" enum="low,medium,high">
  Filter answers by answered helpfulness.
</ParamField>

<ParamField query="relevance" param-type="string[]" enum="relevant,spam,off-topic,unclear">
  Filter answers by question relevance.
</ParamField>

<ParamField query="channelType" param-type="string[]">
  Filter answers by answer channel type.
</ParamField>

<ParamField query="question.type" param-type="string" enum="exploring,how-to,troubleshooting,reference,unknown">
  Filter answers by question type.
</ParamField>

<ParamField query="topic" param-type="string">
  Filter answers by associated site topic ID.
</ParamField>

<ParamField query="thread" param-type="string">
  Filter answers by thread root answer ID. Includes the root answer itself.
</ParamField>

<ParamField query="page" param-type="string">
  Identifier of the page results to fetch.
</ParamField>

<ParamField query="limit" param-type="number">
  The number of results per page
</ParamField>
