Deletes an external link. This operation is intentionally unavailable to MCP clients.

**DELETE** `/orgs/{organizationId}/sites/{siteId}/external-links/{siteExternalLinkId}`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="organizationId" param-type="string" required={true}>
  The unique id of the organization
</ParamField>

<ParamField path="siteId" param-type="string" required={true}>
  The unique id of the site
</ParamField>

<ParamField path="siteExternalLinkId" param-type="string" required={true}>
  The unique ID of the site external link
</ParamField>

**Responses**

`204` External link did not exist

`205` External link has been deleted

`404` Not Found

<ResponseField name="error" field-type="object" required={true}>
  <Expandable title="Object Properties">
    <ResponseField name="code" field-type="integer" required={true} enum="404" />

    <ResponseField name="message" field-type="string" required={true} />
  </Expandable>
</ResponseField>
