Manage Model Context Protocol (Mcp) servers used by your site.

## The SiteMcpServer object

<ResponseField name="object" field-type="string" required={true} enum="site-mcp-server" />

<ResponseField name="id" field-type="string" required={true}>
  Unique identifier for the MCP server
</ResponseField>

<ResponseField name="name" field-type="string" required={true}>
  Name of the MCP server
</ResponseField>

<ResponseField name="url" field-type="string" required={true} />

<ResponseField name="transport" field-type="string" required={true} enum="sse,http">
  Transport protocol used to connect to the MCP server
</ResponseField>

<ResponseField name="headers" field-type="object" required={true}>
  HTTP headers sent with requests to this server
</ResponseField>

<ResponseField name="condition" field-type="string">
  Expression to evaluate
</ResponseField>

<ResponseField name="urls" field-type="object" required={true}>
  URLs associated with the object

  <Expandable title="Object Properties">
    <ResponseField name="location" field-type="string" required={true}>
      URL of the MCP server in the API
    </ResponseField>
  </Expandable>
</ResponseField>
