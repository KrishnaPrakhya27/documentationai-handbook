Modifies an existing MCP server's configuration. Only provided fields are updated; omitting a field leaves it unchanged. Set condition to null to remove it and make the MCP server available to all visitors.

**PATCH** `/orgs/{organizationId}/sites/{siteId}/mcp-servers/{siteMcpServerId}`

**Parameters**

<ParamField header="Authorization" param-type="string" />

<ParamField path="organizationId" param-type="string" required={true}>
  The unique id of the organization
</ParamField>

<ParamField path="siteId" param-type="string" required={true}>
  The unique id of the site
</ParamField>

<ParamField path="siteMcpServerId" param-type="string" required={true}>
  The unique id of the MCP server
</ParamField>

**Request body**

<ParamField body="name" param-type="string">
  Name of the MCP server
</ParamField>

<ParamField body="url" param-type="string" />

<ParamField body="transport" param-type="string" enum="sse,http">
  Transport protocol used to connect to the MCP server
</ParamField>

<ParamField body="headers" param-type="object">
  HTTP headers sent with requests to this server
</ParamField>

<ParamField body="condition" param-type="string">
  Conditional expression used to evaluate whether the MCP server should be available to the site's visitor (should evaluate to a boolean). If not set, the condition will remain unchanged. If set to null, the condition will be removed.
</ParamField>

**Responses**

`200` OK

<ResponseField name="id" field-type="string" required={true}>
  Unique identifier for the MCP server
</ResponseField>

<ResponseField name="name" field-type="string" required={true}>
  Name of the MCP server
</ResponseField>

<ResponseField name="url" field-type="string" required={true} />

<ResponseField name="transport" field-type="string" required={true} enum="sse,http">
  Transport protocol used to connect to the MCP server
</ResponseField>

<ResponseField name="headers" field-type="object" required={true}>
  HTTP headers sent with requests to this server
</ResponseField>

<ResponseField name="condition" field-type="string">
  Expression to evaluate
</ResponseField>

<ResponseField name="urls" field-type="object" required={true}>
  URLs associated with the object

  <Expandable title="Object Properties">

      URL of the MCP server in the API
    </ResponseField>
  </Expandable>
</ResponseField>
