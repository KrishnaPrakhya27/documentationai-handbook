Returns details about the user associated with the authentication provided in the request's authorization header.

**GET** `/user`

**Parameters**

<ParamField header="Authorization" param-type="string" />

**Responses**

`200` OK

<ResponseField name="object" field-type="string" required={true} enum="user">
  Type of Object, always equals to "user"
</ResponseField>

<ResponseField name="id" field-type="string" required={true}>
  Unique identifier for the user
</ResponseField>

<ResponseField name="displayName" field-type="string" required={true}>
  Full name for the user
</ResponseField>

<ResponseField name="email" field-type="string">
  Email address of the user
</ResponseField>

<ResponseField name="photoURL" field-type="string">
  URL of the user's profile picture
</ResponseField>

<ResponseField name="urls" field-type="object" required={true}>
  URLs associated with the object

  <Expandable title="Object Properties">
    <ResponseField name="location" field-type="string" required={true}>
      URL of the user in the API
    </ResponseField>
  </Expandable>
</ResponseField>

`404` Not Found

<ResponseField name="error" field-type="object" required={true}>
  <Expandable title="Object Properties">
    <ResponseField name="code" field-type="integer" required={true} enum="404" />

    <ResponseField name="message" field-type="string" required={true} />
  </Expandable>
</ResponseField>
